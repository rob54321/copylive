# makelive is a perl script to build a live ubuntu system on flash or hard disk.
# the cdrom is copied to LINUXLIVE partition.
The script requires a disk with 4 partitions
partition 1: 8G(default) label=LINUXLIVE
partition 2: 8G label=MCTREC 
partition 3: 10G writable for persistence
partition 4: ele ntfs for rest of disk.
